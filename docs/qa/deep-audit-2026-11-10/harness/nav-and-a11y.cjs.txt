const {chromium}=require('/tmp/kinddesign-tools/node_modules/playwright');const AxeBuilder=require('/tmp/kinddesign-tools/node_modules/@axe-core/playwright').default;const fs=require('node:fs');const assert=require('node:assert/strict');const base='https://kinddesign-holdet-ek-2026.netlify.app';const folder='docs/qa/deep-audit-2026-11-10';
(async()=>{const b=await chromium.launch({channel:'chrome'});const results=[];try{
for(const width of [320,375,390,430,768,1024,1440]){const p=await b.newPage({viewport:{width,height:width<500?568:900},reducedMotion:'no-preference'});for(const route of ['/events/','/om-kinddesign/','/ny-her/','/kontakt/']){await p.goto(base+'/');const nav=width<601?'.mobile-navigation':'.desktop-navigation';await p.locator(nav+` a[href="${route}"]`).click();assert.equal(p.url(),base+route);results.push({id:'nav-journey',width,from:'header',to:route,status:'pass'});await p.goto(base+'/');await p.locator(`footer nav a[href="${route}"]`).click();assert.equal(p.url(),base+route);results.push({id:'nav-journey',width,from:'footer',to:route,status:'pass'});}await p.locator('header .brand').click();assert.equal(p.url(),base+'/');results.push({id:'logo-home',width,status:'pass'});

for(const width of [600,601,640,720,1441,1536,1600,1920]){const p=await b.newPage({viewport:{width,height:900},reducedMotion:'reduce'});for(const route of ['/','/events/','/ny-her/']){await p.goto(base+route);await p.evaluate(()=>document.fonts.ready);const data=await p.evaluate(()=>{const brand=document.querySelector('header .brand').getBoundingClientRect();const nav=[...document.querySelectorAll('header nav')].find(n=>n.getBoundingClientRect().width);const first=nav.querySelector('a').getBoundingClientRect();const cta=nav.querySelector('.new-link').getBoundingClientRect();return{width:innerWidth,scrollWidth:document.documentElement.scrollWidth,logoRight:brand.right,firstLinkLeft:first.left,ctaRight:cta.right,ctaWidth:cta.width,navWidth:nav.getBoundingClientRect().width,gap:getComputedStyle(nav).columnGap}});results.push({id:'extra-header-width',route,width,data});if(data.scrollWidth>width||data.firstLinkLeft<data.logoRight)await p.screenshot({path:`${folder}/findings/header-overflow-${route==='/'?'home':route.replaceAll('/','_')}-${width}.png`});}await p.close();}
fs.writeFileSync(folder+'/nav-observations.json',JSON.stringify(results,null,2));
for(const width of [320,1440]){const context=await b.newContext({viewport:{width,height:900},reducedMotion:'reduce'});const p=await context.newPage();for(const route of ['/','/events/','/events/kalender/','/events/from-ballet-to-technology/','/om-kinddesign/','/ny-her/','/kontakt/']){await p.goto(base+route);const check=await new AxeBuilder({page:p}).withTags(['wcag2a','wcag2aa','wcag21aa','wcag22aa']).analyze();results.push({id:'axe22',width,route,violations:check.violations.map(v=>({id:v.id,impact:v.impact,nodes:v.nodes.map(n=>({target:n.target,summary:n.failureSummary}))})),incomplete:check.incomplete.map(v=>({id:v.id,nodes:v.nodes.length}))});}await p.close();}
fs.writeFileSync(folder+'/nav-and-a11y.json',JSON.stringify(results,null,2));console.log('Navigation and WCAG 2.2 observations saved:',results.length);}finally{await b.close();}})().catch(e=>{console.error(e);process.exitCode=1});
